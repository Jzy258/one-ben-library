# 02 一对多映射

使用 `collection` 标签：
```xml
<resultMap id="departmentResultMap" type="Department">
    <id property="id" column="id"/>
    <result property="dno" column="dno"/>
    <result property="dname" column="dname"/>
    
    <collection property="employees" ofType="Employee">
        <id property="id" column="id"/>
        <result property="eno" column="eno"/>
        <result property="ename" column="ename"/>
    </collection>
</resultMap>
```

分布查询：
```xml
<!-- DepartmentMapper.xml -->
<resultMap 
	id="departmentResultMap" 
	type="cn.ginna.mybatis05.pojo.Department">  
    <id property="id" column="id" />  
    <result property="dno" column="dno" />  
    <result property="dname" column="dname" />  
    <collection 
	    property="employees" 
	    select="cn.ginna.mybatis05.mapper.EmployeeMapper.selectByDno" 
	    column="eno" />  
</resultMap>  
  
<select id="selectByDno" resultMap="departmentResultMap">  
    SELECT  
        dno, dname    
    FROM        
	    t_department    
	WHERE        
		dno = #{dno}
</select>
```
```xml
<!-- EmployeeMapper.xml -->
<resultMap id="employeeResultMap" type="Employee">  
    <id property="id" column="id" />  
    <result property="eno" column="eno" />  
    <result property="ename" column="ename" />  
</resultMap>  
  
<select 
	id="selectByDno" 
	resultType="cn.ginna.mybatis05.pojo.Employee" 
	resultMap="employeeResultMap">  
    SELECT  
        eno, ename    
    FROM        
	    t_employee    
	WHERE        
		dno = #{dno}
</select>
```
